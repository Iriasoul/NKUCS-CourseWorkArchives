`timescale 1ns / 1ps

module tb_week2();
    reg  [199:0] img_in;
    reg  [71:0]  weight_in;
    wire [31:0]  cnn_out;

    cnn_core_top u_top (
        .img_5x5(img_in),
        .weight_3x3(weight_in),
        .out_2x2(cnn_out)
    );

    initial begin
        // Test1：标准5x5灰度矩阵，像素全10，权重全1
        weight_in = {9{8'd1}};
        img_in    = {25{8'd10}}; 
        #100;
        $display("Test1 | weight: %h | img: %h | out: %d, %d, %d, %d", 
                 weight_in, img_in, cnn_out[31:24], cnn_out[23:16], cnn_out[15:8], cnn_out[7:0]);
        
        // Test2：孤点图像，仅像素0为20，其余为0，权重全1
        img_in = 200'b0; 
        img_in[7:0] = 8'd20; 
        #100;
        $display("Test2 | weight: %h | img: %h | out: %d, %d, %d, %d", 
                 weight_in, img_in, cnn_out[31:24], cnn_out[23:16], cnn_out[15:8], cnn_out[7:0]);
    
        // Test3：零输入，图像全0，权重全1
        img_in = 200'b0;
        #100;
        $display("Test3 | weight: %h | img: %h | out: %d, %d, %d, %d", 
                 weight_in, img_in, cnn_out[31:24], cnn_out[23:16], cnn_out[15:8], cnn_out[7:0]);
        
        // Test4：低亮度，像素全4，权重全1
        img_in = {25{8'd4}};
        #100;
        $display("Test4 | weight: %h | img: %h | out: %d, %d, %d, %d", 
                 weight_in, img_in, cnn_out[31:24], cnn_out[23:16], cnn_out[15:8], cnn_out[7:0]);
        
        // Test5：提升权重为2，像素全10
        img_in    = {25{8'd10}};
        weight_in = {9{8'd2}};
        #100;
        $display("Test5 | weight: %h | img: %h | out: %d, %d, %d, %d", 
                 weight_in, img_in, cnn_out[31:24], cnn_out[23:16], cnn_out[15:8], cnn_out[7:0]);
        
        // Test6：对角线，对角线的像素为10，权重全1
        img_in = 200'b0;
        img_in[7:0]    = 8'd10;
        img_in[55:48]  = 8'd10;
        img_in[103:96] = 8'd10;
        img_in[151:144]=8'd10;
        img_in[199:192]=8'd10;
        weight_in = {9{8'd1}};
        #100;
        $display("Test6 | weight: %h | img: %h | out: %d, %d, %d, %d", 
                 weight_in, img_in, cnn_out[31:24], cnn_out[23:16], cnn_out[15:8], cnn_out[7:0]);
        
        // Test7：权重全0，像素全255，
        img_in    = {25{8'd255}};
        weight_in = 72'b0;
        #100;
        $display("Test7 | weight: %h | img: %h | out: %d, %d, %d, %d", 
                 weight_in, img_in, cnn_out[31:24], cnn_out[23:16], cnn_out[15:8], cnn_out[7:0]);
        
        $finish;
    end
endmodule
