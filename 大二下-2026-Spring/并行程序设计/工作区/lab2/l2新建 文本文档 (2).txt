template<typename T>
T *LoadData(std::string data_path, size_t& n, size_t& d)
{
    std::ifstream fin;
    fin.open(data_path, std::ios::in | std::ios::binary);
    if(!fin.is_open()) {
        std::cerr << "Failed to open " << data_path << std::endl;
        exit(-1);
    }
    fin.read((char*)&n,4);
    fin.read((char*)&d,4);
    T* data = new T[n*d];
    int sz = sizeof(T);
    for(size_t i = 0; i < n; ++i){
        fin.read(((char*)data + i*d*sz), d*sz);
    }
    fin.close();

    std::cerr<<"load data "<<data_path<<"\n";
    std::cerr<<"dimension: "<<d<<"  number:"<<n<<"  size_per_element:"<<sizeof(T)<<"\n";

    return data;
}

struct SearchResult
{
    float recall;
    int64_t latency; // 单位us
};

// 1. Flat SIMD 优化
std::priority_queue<std::pair<float, uint32_t> > flat_search_simd(float* base, float* query, size_t base_number, size_t vecdim, size_t k) {
    std::priority_queue<std::pair<float, uint32_t> > q;

    for(size_t i = 0; i < base_number; ++i) {
        float32x4_t sum_vec = vdupq_n_f32(0.0f);
        float* current_base = base + i * vecdim;
        
        // 使用 NEON 指令进行向量化内积计算，一次处理 4 个 float
        size_t d = 0;
        for(; d + 3 < vecdim; d += 4) {
            float32x4_t a = vld1q_f32(current_base + d);
            float32x4_t b = vld1q_f32(query + d);
            sum_vec = vfmaq_f32(sum_vec, a, b);
        }
        
        float sum_arr[4];
        vst1q_f32(sum_arr, sum_vec);
        float dis = sum_arr[0] + sum_arr[1] + sum_arr[2] + sum_arr[3];
        
        // 处理剩余的维度（这里其实不需要，96维正好是4的倍数）
        for(; d < vecdim; ++d) {
            dis += current_base[d] * query[d];
        }
        
        dis = 1.0f - dis;

        if(q.size() < k) {
            q.push({dis, i});
        } else {
            if(dis < q.top().first) {
                q.push({dis, i});
                q.pop();
            }
        }
    }
    return q;
}

// 2. SQ SIMD 优化
uint8_t* sq_base = nullptr;
float sq_min = 0, sq_max = 0, sq_scale = 1.0f;

// SQ 预处理
void offline_sq(float* base, size_t base_number, size_t vecdim) {
    sq_base = new uint8_t[base_number * vecdim];
    sq_min = base[0];
    sq_max = base[0];
    for (size_t i = 0; i < base_number * vecdim; ++i) {
        if (base[i] < sq_min) sq_min = base[i];
        if (base[i] > sq_max) sq_max = base[i];
    }
    sq_scale = 255.0f / (sq_max - sq_min);
    
    #pragma omp parallel for
    for (size_t i = 0; i < base_number * vecdim; ++i) {
        sq_base[i] = static_cast<uint8_t>((base[i] - sq_min) * sq_scale);
    }
}

// SQ 查询
std::priority_queue<std::pair<float, uint32_t> > sq_search_simd(uint8_t* base_u8, float* query, size_t base_number, size_t vecdim, size_t k) {
    std::priority_queue<std::pair<float, uint32_t> > q;
    
    // 对 query 进行量化
    uint8_t query_u8[vecdim];
    for(size_t d = 0; d < vecdim; ++d) {
        query_u8[d] = static_cast<uint8_t>((std::max(sq_min, std::min(sq_max, query[d])) - sq_min) * sq_scale);
    }

    for(size_t i = 0; i < base_number; ++i) {
        uint8_t* current_base = base_u8 + i * vecdim;
        uint32x4_t sum_vec = vdupq_n_u32(0);
        
        size_t d = 0;
        // 一次处理 16 个 uint8
        for(; d + 15 < vecdim; d += 16) {
            uint8x16_t a = vld1q_u8(current_base + d);
            uint8x16_t b = vld1q_u8(query_u8 + d);
            
            // 扩展计算，防止溢出
            uint16x8_t a_low = vmovl_u8(vget_low_u8(a));
            uint16x8_t a_high = vmovl_u8(vget_high_u8(a));
            uint16x8_t b_low = vmovl_u8(vget_low_u8(b));
            uint16x8_t b_high = vmovl_u8(vget_high_u8(b));
            
            uint32x4_t mul1 = vmull_u16(vget_low_u16(a_low), vget_low_u16(b_low));
            uint32x4_t mul2 = vmull_u16(vget_high_u16(a_low), vget_high_u16(b_low));
            uint32x4_t mul3 = vmull_u16(vget_low_u16(a_high), vget_low_u16(b_high));
            uint32x4_t mul4 = vmull_u16(vget_high_u16(a_high), vget_high_u16(b_high));
            
            sum_vec = vaddq_u32(sum_vec, mul1);
            sum_vec = vaddq_u32(sum_vec, mul2);
            sum_vec = vaddq_u32(sum_vec, mul3);
            sum_vec = vaddq_u32(sum_vec, mul4);
        }
        
        uint32_t sum_arr[4];
        vst1q_u32(sum_arr, sum_vec);
        uint32_t dis_int = sum_arr[0] + sum_arr[1] + sum_arr[2] + sum_arr[3];
        
        for(; d < vecdim; ++d) {
            dis_int += current_base[d] * query_u8[d];
        }

        // 近似恢复到原量级作为距离比较（SQ 只要单调即可）
        // 实际应用中可以只比较整型以提高速度，这里为了和 Flat 对齐输出保持浮点
        float dis = 1.0f - (float)dis_int / (sq_scale * sq_scale) - sq_min * sq_min * vecdim;

        if(q.size() < k) {
            q.push({dis, i});
        } else {
            if(dis < q.top().first) {
                q.push({dis, i});
                q.pop();
            }
        }
    }
    return q;
}

// 3. PQ SIMD 优化
const int PQ_M = 4; // 子段数量
const int PQ_K = 256; // 聚类中心数 (8-bit)
uint8_t* pq_codes = nullptr;
float* pq_centroids = nullptr; // 存储结构: M * K * (vecdim/M)

// 计算子段距离
float calc_sub_dist(float* a, float* b, size_t sub_dim) {
    float dis = 0;
    for(size_t i=0; i<sub_dim; ++i) dis += a[i] * b[i];
    return dis;
}

// 简易 KMeans，用于 PQ 离线预处理
void simple_kmeans(float* data, size_t n, size_t dim, float* centroids) {
    // 随机初始化
    for(int i=0; i<PQ_K; ++i) {
        int idx = rand() % n;
        std::memcpy(centroids + i*dim, data + idx*dim, dim * sizeof(float));
    }
    int iters = 10;
    std::vector<int> assign(n);
    std::vector<int> counts(PQ_K);
    std::vector<float> new_cents(PQ_K * dim);

    while(iters--) {
        #pragma omp parallel for
        for(size_t i=0; i<n; ++i) {
            float min_d = -1e9; // 由于是内积，越大越好
            int best_c = 0;
            for(int c=0; c<PQ_K; ++c) {
                float d = calc_sub_dist(data + i*dim, centroids + c*dim, dim);
                if(d > min_d) { min_d = d; best_c = c; }
            }
            assign[i] = best_c;
        }
        
        std::fill(new_cents.begin(), new_cents.end(), 0.0f);
        std::fill(counts.begin(), counts.end(), 0);
        for(size_t i=0; i<n; ++i) {
            int c = assign[i];
            counts[c]++;
            for(size_t d=0; d<dim; ++d) new_cents[c*dim + d] += data[i*dim + d];
        }
        for(int c=0; c<PQ_K; ++c) {
            if(counts[c] > 0) {
                for(size_t d=0; d<dim; ++d) centroids[c*dim + d] = new_cents[c*dim + d] / counts[c];
            }
        }
    }
}

// PQ 预处理
void offline_pq(float* base, size_t base_number, size_t vecdim) {
    size_t sub_dim = vecdim / PQ_M;
    pq_codes = new uint8_t[base_number * PQ_M];
    pq_centroids = new float[PQ_M * PQ_K * sub_dim];
    
    std::vector<float> sub_data(base_number * sub_dim);
    
    for(int m=0; m<PQ_M; ++m) {
        // 收集子空间数据
        for(size_t i=0; i<base_number; ++i) {
            std::memcpy(&sub_data[i*sub_dim], base + i*vecdim + m*sub_dim, sub_dim * sizeof(float));
        }
        // 训练 KMeans
        simple_kmeans(sub_data.data(), base_number, sub_dim, pq_centroids + m*PQ_K*sub_dim);
        
        // 编码 Base
        #pragma omp parallel for
        for(size_t i=0; i<base_number; ++i) {
            float max_ip = -1e9;
            int best_c = 0;
            for(int c=0; c<PQ_K; ++c) {
                float ip = calc_sub_dist(base + i*vecdim + m*sub_dim, pq_centroids + m*PQ_K*sub_dim + c*sub_dim, sub_dim);
                if(ip > max_ip) { max_ip = ip; best_c = c; }
            }
            pq_codes[i*PQ_M + m] = best_c;
        }
    }
}

// PQ查询 (ADC)
std::priority_queue<std::pair<float, uint32_t> > pq_search_simd(float* query, size_t base_number, size_t vecdim, size_t k) {
    size_t sub_dim = vecdim / PQ_M;
    float lut[PQ_M][PQ_K]; // 局部查找表
    
    // 构建 LUT (ADC)
    for(int m=0; m<PQ_M; ++m) {
        for(int c=0; c<PQ_K; ++c) {
            lut[m][c] = calc_sub_dist(query + m*sub_dim, pq_centroids + m*PQ_K*sub_dim + c*sub_dim, sub_dim);
        }
    }

    std::priority_queue<std::pair<float, uint32_t> > q;
    
    // 查表累计距离
    for(size_t i=0; i<base_number; ++i) {
        float dis = 0;
        uint8_t* code = pq_codes + i*PQ_M;
        for(int m=0; m<PQ_M; ++m) {
            dis += lut[m][code[m]];
        }
        dis = 1.0f - dis; // 转回距离度量

        if(q.size() < k) {
            q.push({dis, i});
        } else {
            if(dis < q.top().first) {
                q.push({dis, i});
                q.pop();
            }
        }
    }
    return q;
}




int main(int argc, char *argv[])
{
    size_t test_number = 0, base_number = 0;
    size_t test_gt_d = 0, vecdim = 0;

    std::string data_path = "/anndata/"; 
    auto test_query = LoadData<float>(data_path + "DEEP100K.query.fbin", test_number, vecdim);
    auto test_gt = LoadData<int>(data_path + "DEEP100K.gt.query.100k.top100.bin", test_number, test_gt_d);
    auto base = LoadData<float>(data_path + "DEEP100K.base.100k.fbin", base_number, vecdim);
    
    // 只测试前2000条查询
    test_number = 2000;
    const size_t k = 10;
    std::vector<SearchResult> results;
    results.resize(test_number);

    // 对比不同算法: 0 = Flat, 1 = Flat-SIMD, 2 = SQ-SIMD, 3 = PQ-SIMD
    int strategy = 1; 

    std::cout << "Strategy selected: " << strategy << std::endl;

    // 离线预处理
    if (strategy == 2) {
        std::cout << "Building SQ index offline..." << std::endl;
        offline_sq(base, base_number, vecdim);
    } else if (strategy == 3) {
        std::cout << "Building PQ index offline..." << std::endl;
        offline_pq(base, base_number, vecdim);
    }

    // 查询测试
    for(size_t i = 0; i < test_number; ++i) {
        const unsigned long Converter = 1000 * 1000;
        struct timeval val;
        int ret = gettimeofday(&val, NULL);

        std::priority_queue<std::pair<float, uint32_t>> res;

        // 使用不同算法进行查询
        if (strategy == 0) {
            res = flat_search(base, test_query + i*vecdim, base_number, vecdim, k);
        } else if (strategy == 1) {
            res = flat_search_simd(base, test_query + i*vecdim, base_number, vecdim, k);
        } else if (strategy == 2) {
            res = sq_search_simd(sq_base, test_query + i*vecdim, base_number, vecdim, k);
        } else if (strategy == 3) {
            res = pq_search_simd(test_query + i*vecdim, base_number, vecdim, k);
        }

        struct timeval newVal;
        ret = gettimeofday(&newVal, NULL);
        int64_t diff = (newVal.tv_sec * Converter + newVal.tv_usec) - (val.tv_sec * Converter + val.tv_usec);

        std::set<uint32_t> gtset;
        for(size_t j = 0; j < k; ++j){
            int t = test_gt[j + i*test_gt_d];
            gtset.insert(t);
        }

        size_t acc = 0;
        while (!res.empty()) {   
            int x = res.top().second;
            if(gtset.find(x) != gtset.end()){
                ++acc;
            }
            res.pop();
        }
        float recall = (float)acc/k;

        results[i] = {recall, diff};
    }

    float avg_recall = 0, avg_latency = 0;
    for(size_t i = 0; i < test_number; ++i) {
        avg_recall += results[i].recall;
        avg_latency += results[i].latency;
    }

    std::cout << "average recall: "<<avg_recall / test_number<<"\n";
    std::cout << "average latency (us): "<<avg_latency / test_number<<"\n";
    
    // 释放内存
    delete[] test_query;
    delete[] test_gt;
    delete[] base;
    if(sq_base) delete[] sq_base;
    if(pq_codes) delete[] pq_codes;
    if(pq_centroids) delete[] pq_centroids;

    return 0;
}
